## ◈ `whoami`

```python
# ─── battery_dna_plus.profile ────────────────────────────

class BatteryDNAPlus:

    PROBLEM = """
    A used EV battery can look perfectly normal on the outside
    while its internal condition remains unknown.
    A QR code or serial number shows IDENTITY, not CONDITION.
    """

    SOLUTION = """
    Analyze measurable behavior (voltage, current, temperature,
    resistance and BMS data) to build a Battery DNA profile,
    detect anomalies, estimate a Trust Score and recommend
    the battery's next life.
    """

    INPUTS = [
        "Voltage", "Current", "Temperature",
        "Internal Resistance / Impedance", "BMS Data",
    ]

    OUTPUTS = [
        "Battery DNA Profile", "Battery Trust Score",
        "Tamper / Anomaly Risk", "Future Degradation Trend",
        "CONTINUE | REPAIR | REPURPOSE | RECYCLE",
    ]

    CORE_IDEA = "Battery Identity != Battery Condition"
    DISCLAIMER = "AI-assisted assessment. No guarantee of authenticity or safety."
```
